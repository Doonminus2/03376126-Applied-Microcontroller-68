1.4 ขั้นตอนการทดลองทีละขั้น

  1. เริ่มต้น NVS Flash: เรียก nvs_flash_init() เนื่องจากเลเยอร์ PHY ของ Wi-Fi อ่านข้อมูลการปรับเทียบ (calibration) จาก NVS หากหน้าหน่วยความจำเสียหาย ให้จัดการด้วย nvs_flash_erase() ตามความจำเป็น
  2. เริ่มต้นสแต็กเครือข่าย TCP/IP: เรียก esp_netif_init() และสร้าง event loop เริ่มต้นด้วย esp_event_loop_create_default()
  3. สร้างอินเทอร์เฟซเครือข่าย STA: เรียก esp_netif_create_default_wifi_sta()
  4. ลงทะเบียน Event Callback: ลงทะเบียน handler ของ WIFI_EVENT และ IP_EVENT เพื่อจัดการการเปลี่ยนสถานะ (เช่น WIFI_EVENT_STA_START, IP_EVENT_STA_GOT_IP)
  5. ตั้งค่าและเริ่มไดรเวอร์ Wi-Fi: กำหนดค่า wifi_config_t ด้วย SSID "Wokwi-GUEST" แล้วเริ่มสแต็กไดรเวอร์
  6. ซิงโครไนซ์การทำงาน: บล็อกการทำงานไว้ที่ xEventGroupWaitBits() จนกว่าจะได้รับ IP แบบไดนามิก (DHCP)

1.5 ซอร์สโค้ดภาษา C ฉบับสมบูรณ์ (main/main.c)

#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/event_groups.h"
#include "esp_system.h"
#include "esp_wifi.h"
#include "esp_event.h"
#include "esp_log.h"
#include "nvs_flash.h"

#define WIFI_SSID "Wokwi-GUEST"
#define WIFI_PASS ""

static const char *TAG = "WIFI_PROJECT";
static EventGroupHandle_t s_wifi_event_group;
#define WIFI_CONNECTED_BIT BIT0

static void event_handler(void* arg, esp_event_base_t event_base, int32_t event_id, void* event_data) {
    if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {
        esp_wifi_connect();
    } else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_DISCONNECTED) {
        ESP_LOGW(TAG, "Retrying Wi-Fi connection...");
        esp_wifi_connect();
    } else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t* event = (ip_event_got_ip_t*) event_data;
        ESP_LOGI(TAG, "Successfully acquired IP address: " IPSTR, IP2STR(&event->ip_info.ip));
        xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
    }
}

void app_main(void) {
    // 1. Initialize NVS Flash Memory
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);

    // 2. Initialize Core Event Groups & Network Stack
    s_wifi_event_group = xEventGroupCreate();
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_sta();

    // 3. Initialize Wi-Fi Driver
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    // 4. Register System Event Loop Callbacks
    esp_event_handler_instance_t instance_any_id;
    esp_event_handler_instance_t instance_got_ip;
    ESP_ERROR_CHECK(esp_event_handler_instance_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &event_handler, NULL, &instance_any_id));
    ESP_ERROR_CHECK(esp_event_handler_instance_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &event_handler, NULL, &instance_got_ip));

    // 5. Configure Wi-Fi Station Credentials
    wifi_config_t wifi_config = {
        .sta = {
            .ssid = WIFI_SSID,
            .password = WIFI_PASS,
        },
    };
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "Wi-Fi initialization finished. Waiting for connection...");
    xEventGroupWaitBits(s_wifi_event_group, WIFI_CONNECTED_BIT, pdFALSE, pdTRUE, portMAX_DELAY);
    ESP_LOGI(TAG, "Network pipeline ready for HTTP/MQTT traffic!");
}

1.6 รายการตรวจสอบเพื่อประเมินตนเอง

หัวข้อการประเมิน ผลลัพธ์ที่ต้องตรวจสอบ สถานะ

2.4 ซอร์สโค้ดภาษา C ฉบับสมบูรณ์ (main/main.c)

#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/i2c_master.h"
#include "esp_log.h"

#define I2C_SCL_IO           GPIO_NUM_22
#define I2C_SDA_IO           GPIO_NUM_21
#define I2C_MASTER_FREQ_HZ   100000
#define LCD_SLAVE_ADDR       0x27

static const char *TAG = "I2C_PROJECT";

void app_main(void) {
    ESP_LOGI(TAG, "Initializing Modern I2C Master Bus...");

    // 1. Define Bus Configuration Structure
    i2c_master_bus_config_t bus_config = {
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .i2c_port = I2C_NUM_0,
        .scl_io_num = I2C_SCL_IO,
        .sda_io_num = I2C_SDA_IO,
        .glitch_ignore_cnt = 7,
        .flags.enable_internal_pullup = true,
    };

    // 2. Instantiate Master Bus Handle
    i2c_master_bus_handle_t bus_handle;
    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_config, &bus_handle));

    // 3. Register Slave Device Configuration
    i2c_device_config_t dev_config = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address = LCD_SLAVE_ADDR,
        .scl_speed_hz = I2C_MASTER_FREQ_HZ,
    };

    i2c_master_dev_handle_t dev_handle;
    ESP_ERROR_CHECK(i2c_master_bus_add_device(bus_handle, &dev_config, &dev_handle));

    ESP_LOGI(TAG, "I2C bus registered successfully. Probing slave device at address 0x27...");

    // 4. Perform Data Transmission Probe
    uint8_t init_cmd[] = { 0x08 }; // Backlight control / clear command frame
    esp_err_t err = i2c_master_transmit(dev_handle, init_cmd, sizeof(init_cmd), 1000 / portTICK_PERIOD_MS);

    if (err == ESP_OK) {
        ESP_LOGI(TAG, "Communication success! Peripheral acknowledged transfer (ACK).");
    } else {
        ESP_LOGE(TAG, "Communication failed: %s", esp_err_to_name(err));
    }
}

2.5 รายการตรวจสอบเพื่อประเมินตนเอง

หัวข้อการประเมิน ผลลัพธ์ที่ต้องตรวจสอบ สถานะ
การเริ่มต้น I2C Master i2c_new_master_bus() ทำงานได้เรียบร้อยและคืนค่า ESP_OK PASS
การลงทะเบียนอุปกรณ์ Slave Device handle ผูกกับที่อยู่ 7 บิต 0x27 ที่ความเร็วสัญญาณนาฬิกา 100 kHz PASS
การตรวจสอบการตอบกลับ ACK i2c_master_transmit() สำเร็จโดยไม่มีข้อผิดพลาด timeout ของฮาร์ดแวร์ PASS

3. โมดูลที่ 3: RTOS และสถาปัตยกรรมระบบ — การประมวลผลอินเทอร์รัปต์แบบเลื่อนเวลาและคิว

3.1 วัตถุประสงค์และความรู้พื้นฐาน

ฝึกการทำงานแบบเรียลไทม์ที่คาดการณ์ได้ (deterministic) โดยหลีกเลี่ยงการวนตรวจสอบ (polling) ที่ใช้เวลานานในลูปหลัก สร้างอินเทอร์รัปต์ฮาร์ดแวร์แบบทริกเกอร์ที่ขอบสัญญาณ (GPIO ISR) รันโค้ดในบริบทอินเทอร์รัปต์อย่างปลอดภัยด้วยแอตทริบิวต์ IRAM_ATTR ส่งต่องานประมวลผลอย่างปลอดภัยผ่าน FreeRTOS Queue (xQueueSendFromISR) และปลดบล็อก task ทำงานเบื้องหลังแบบอะซิงโครนัส

#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/queue.h"
#include "driver/gpio.h"
#include "esp_log.h"

#define BUTTON_PIN GPIO_NUM_4

static const char *TAG = "RTOS_ISR_PROJECT";
static QueueHandle_t gpio_evt_queue = NULL;

// 1. Interrupt Service Routine placed in internal Instruction RAM (IRAM)
static void IRAM_ATTR gpio_isr_handler(void* arg) {
    uint32_t gpio_num = (uint32_t) arg;
    xQueueSendFromISR(gpio_evt_queue, &gpio_num, NULL);
}

// 2. Background Worker Consumer Thread
static void button_worker_task(void* arg) {
    uint32_t io_num;
    uint32_t press_count = 0;
    while (1) {
        // Block indefinitely until an ISR event is pushed to the queue
        if (xQueueReceive(gpio_evt_queue, &io_num, portMAX_DELAY)) {
            press_count++;
            ESP_LOGI(TAG, "ISR Event Received! Button on GPIO %ld pressed %ld times.", io_num, press_count);
        }
    }
}

void app_main(void) {
    // 3. Create FreeRTOS Queue for Event Offloading
    gpio_evt_queue = xQueueCreate(10, sizeof(uint32_t));

    // 4. Spawn Consumer Task
    xTaskCreate(button_worker_task, "button_worker_task", 2048, NULL, 10, NULL);

    // 5. Configure Hardware Edge Trigger on Pin
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << BUTTON_PIN),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_NEGEDGE,
    };
    gpio_config(&io_conf);

    // 6. Install Global ISR Service and Register Handler
    gpio_install_isr_service(0);
    gpio_isr_handler_add(BUTTON_PIN, gpio_isr_handler, (void*) BUTTON_PIN);

    ESP_LOGI(TAG, "GPIO Interrupt Service Routine installed. Waiting for hardware button presses...");
}
กฎสำคัญของ ISR

ฟังก์ชันที่ทำงานภายใน ISR ห้ามเรียกคำสั่งที่บล็อกการทำงาน เช่น vTaskDelay(), printf() หรือฟังก์ชัน RTOS ที่ไม่ใช่เวอร์ชันสำหรับ ISR โดยเด็ดขาด ฟังก์ชัน RTOS ทุกตัวที่เรียกจาก ISR ต้องลงท้ายด้วย FromISR

3.5 รายการตรวจสอบเพื่อประเมินตนเอง

หัวข้อการประเมิน ผลลัพธ์ที่ต้องตรวจสอบ สถานะ
ตำแหน่งการทำงานของ ISR ฟังก์ชัน handler ระบุ IRAM_ATTR ไว้อย่างชัดเจน PASS
การส่งต่องานผ่านคิวอย่างปลอดภัย xQueueSendFromISR() ส่งหมายเลขขาเข้าคิวได้เรียบร้อยโดยไม่เกิดการแครช PASS
การปลดบล็อก Task Task ทำงานเบื้องหลังประมวลผลข้อความที่เข้ามาแบบอะซิงโครนัส PASS

4. โมดูลที่ 4: ระบบไฟล์บนแฟลช — การจัดเก็บแบบ Wear-Leveling ด้วย SPIFFS และ LittleFS

4.1 วัตถุประสงค์และความรู้พื้นฐาน

เข้าใจการจัดเก็บข้อมูลแบบมีโครงสร้างที่ไม่สูญหายเมื่อไฟดับบนหน่วยความจำแฟลชแบบ NOR สร้างพาร์ทิชัน SPIFFS / LittleFS โดยใช้เลเยอร์ Virtual File System (VFS) ของ Espressif เพื่อทำงานกับไฟล์ด้วยคำสั่ง POSIX มาตรฐาน (fopen, fprintf, fgets, fclose) พร้อมใช้ประโยชน์จากอัลกอริทึม wear-leveling

4.4 ซอร์สโค้ดภาษา C ฉบับสมบูรณ์ (main/main.c)

#include <stdio.h>
#include <string.h>
#include "esp_spiffs.h"
#include "esp_log.h"

static const char *TAG = "STORAGE_PROJECT";

void app_main(void) {
    ESP_LOGI(TAG, "Initializing SPIFFS Virtual File System...");

    // 1. Configure VFS SPIFFS Mounting Structure
    esp_vfs_spiffs_conf_t conf = {
        .base_path = "/spiffs",
        .partition_label = NULL,
        .max_files = 5,
        .format_if_mount_failed = true
    };

    // 2. Register and Mount VFS SPIFFS
    esp_err_t ret = esp_vfs_spiffs_register(&conf);
    if (ret != ESP_OK) {
        if (ret == ESP_FAIL) {
            ESP_LOGE(TAG, "Failed to mount or format filesystem");
        } else if (ret == ESP_ERR_NOT_FOUND) {
            ESP_LOGE(TAG, "Failed to find SPIFFS partition in partition table");
        } else {
            ESP_LOGE(TAG, "Failed to initialize SPIFFS (%s)", esp_err_to_name(ret));
        }
        return;
    }

    ESP_LOGI(TAG, "SPIFFS mounted successfully at /spiffs. Executing POSIX Write Operation...");

    // 3. Perform Standard POSIX File Write Call
    FILE* f = fopen("/spiffs/log.txt", "w");
    if (f == NULL) {
        ESP_LOGE(TAG, "Failed to open log.txt for writing");
        return;
    }
    fprintf(f, "Sensor Log Entry #104: Temp=24.5C, Humidity=60%%
");
    fclose(f);
    ESP_LOGI(TAG, "File successfully written and flushed to flash.");

    // 4. Perform POSIX Read and Verification
    FILE* f_read = fopen("/spiffs/log.txt", "r");
    if (f_read == NULL) {
        ESP_LOGE(TAG, "Failed to open log.txt for reading");
        return;
    }
    char line[128];
    fgets(line, sizeof(line), f_read);
    fclose(f_read);

    ESP_LOGI(TAG, "Read back string from SPIFFS: %s", line);

    // 5. Unmount Filesystem Cleanly
    esp_vfs_spiffs_unregister(conf.partition_label);
    ESP_LOGI(TAG, "SPIFFS filesystem unmounted cleanly.");
}

4.5 รายการตรวจสอบเพื่อประเมินตนเอง

หัวข้อการประเมิน ผลลัพธ์ที่ต้องตรวจสอบ สถานะ
การลงทะเบียนพาร์ทิชัน esp_vfs_spiffs_register() คืนค่า ESP_OK โดยไม่มีการฟอร์แมตล้มเหลว PASS
การเขียนไฟล์ด้วย POSIX fopen() และ fprintf() บันทึกบัฟเฟอร์ลงแฟลชได้สำเร็จ PASS
การตรวจสอบความถูกต้องของข้อมูล ข้อมูลที่อ่านกลับมาตรงกับสตริงที่เขียนทุกตัวอักษร PASS

5. การประเมินตนเองแบบรวมและเกณฑ์การให้คะแนน (Rubric)

ประเมินความเชี่ยวชาญทางเทคนิคโดยรวมของคุณทั้ง 4 โมดูลการจำลอง ESP32 ตามเกณฑ์ด้านล่าง:

ด้านการเรียนรู้ สมรรถนะทางเทคนิคหลัก คะแนน การสะท้อนตนเองและหลักฐานจากระบบ
1. สแต็กโปรโตคอลเครือข่าย การตั้งค่า esp_netif, การจัดการการเชื่อมต่อ Wi-Fi แบบขับเคลื่อนด้วยอีเวนต์ และการรับ IP แบบไดนามิกจาก DHCP 5 / 5 เชื่อมต่อ STA กับ AP Wokwi-GUEST ได้ และปลดบล็อก FreeRTOS Event Group เมื่อเกิด IP_EVENT_STA_GOT_IP
2. ไดรเวอร์บัสฮาร์ดแวร์ การสร้าง bus handle ด้วย driver/i2c_master.h ของ ESP-IDF v5 และการรับส่งข้อมูลกับอุปกรณ์ slave 5 / 5 ตั้งค่าบัส I2C master บน GPIO 21/22 และได้รับการยืนยัน ACK จากคอนโทรลเลอร์ PCF8574 ที่ที่อยู่ 0x27
3. รูปแบบการใช้ระบบปฏิบัติการเวลาจริง การแยก ISR ของฮาร์ดแวร์ออกจาก task ผู้บริโภคข้อมูลด้วย handler ใน IRAM และ FreeRTOS Queue 5 / 5 ไม่มีลูปที่บล็อกการทำงาน ISR ทำงานจาก IRAM เมื่อเกิดขอบสัญญาณ และส่งข้อมูลผ่านคิวที่ปลอดภัยต่อเธรด
4. ระบบจัดเก็บข้อมูลแบบฝังตัว การแบ่งพาร์ทิชันแฟลชเอง การเมานต์เลเยอร์ VFS และการทำงานกับไฟล์ด้วยคำสั่ง POSIX มาตรฐานของภาษา C 5 / 5 เมานต์พาร์ทิชัน SPIFFS ได้เรียบร้อย บันทึกสตริงข้อมูลเซนเซอร์ และตรวจสอบการอ่านกลับได้ตรงทุกไบต์

แบบทดสอบความเข้าใจท้ายบท

  • [✔] ทำไมต้องเริ่มต้น NVS flash ก่อนเริ่มสแต็ก Wi-Fi?
    คำตอบ: เลเยอร์ PHY ของ Wi-Fi ของ Espressif ใช้เซกเตอร์ใน NVS เพื่อดึงข้อมูลการปรับเทียบคลื่นวิทยุ (RF calibration) การตั้งค่า MAC address และกู้คืนพารามิเตอร์การทำงานของ Wi-Fi
  • [✔] สถาปัตยกรรมของ driver/i2c.h แบบเดิม ต่างจาก driver/i2c_master.h แบบใหม่อย่างไร?
    คำตอบ: ไดรเวอร์ใหม่ของ ESP-IDF v5.x ใช้ bus handle (i2c_master_bus_handle_t) และ device handle (i2c_master_dev_handle_t) อย่างชัดเจน ทำให้ใช้งาน slave หลายตัวได้อย่างปลอดภัยต่อเธรด และรองรับการจัดการพลังงาน
  • [✔] ทำไมจึงห้ามเรียกคำสั่งที่บล็อก (เช่น vTaskDelay, printf) ภายในฟังก์ชัน ISR ที่ระบุ IRAM_ATTR?
    คำตอบ: ISR ทำงานในบริบทอินเทอร์รัปต์ซึ่งมีลำดับความสำคัญของ CPU สูง การบล็อกหรือการรันโค้ดที่อยู่ในแคชของแฟลชจะทำให้การจัดลำดับงาน (scheduling) ของ CPU หยุดชะงัก และเสี่ยงต่อการเกิด panic จาก cache miss ขณะที่อินเทอร์รัปต์ถูกปิดกั้นอยู่
  • [✔] Virtual File System (VFS) แปลงคำสั่ง POSIX มาตรฐาน (fopen) ไปยังหน่วยความจำแฟลชได้อย่างไร?
    คำตอบ: VFS ลงทะเบียนพอยน์เตอร์ฟังก์ชันของไดรเวอร์ (read/write/open) ที่ผูกกับไดรเวอร์ระบบไฟล์จริงด้านล่าง (SPIFFS/LittleFS) ซึ่งแปลง path เสมือนเป็นคำสั่งลบเซกเตอร์และ wear-leveling บนแฟลช
